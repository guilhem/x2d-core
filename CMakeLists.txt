cmake_minimum_required(VERSION 3.16)
project(x2d_core VERSION 0.1.0 LANGUAGES CXX)

add_library(x2d_core INTERFACE)
target_compile_features(x2d_core INTERFACE cxx_std_17)
target_include_directories(x2d_core INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/src)

include(CTest)
if(BUILD_TESTING)
  foreach(check IN ITEMS radio radio_runtime queue cc1101 journal_fixture controller mysensors)
    add_executable(check_${check} tests/check_${check}.cpp)
    target_link_libraries(check_${check} PRIVATE x2d_core)
    target_compile_options(check_${check} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    add_test(NAME ${check} COMMAND check_${check})
  endforeach()
  target_compile_definitions(check_journal_fixture PRIVATE
    JOURNAL_FIXTURE="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/journal-v1-a0f98d0.bin")
  if(UNIX)
    add_executable(mysensors_server tests/mysensors_server.cpp)
    target_link_libraries(mysensors_server PRIVATE x2d_core)
    target_compile_options(mysensors_server PRIVATE -Wall -Wextra -Werror)
  endif()
endif()
