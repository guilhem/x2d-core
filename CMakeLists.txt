cmake_minimum_required(VERSION 3.16)
project(x2d_core VERSION 0.1.0 LANGUAGES CXX)

add_library(x2d_core INTERFACE)
target_compile_features(x2d_core INTERFACE cxx_std_17)
target_include_directories(x2d_core INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/src)

# Only the JSONL gateway needs ArduinoJson; the radio/journal target stays plain C++.
set(ARDUINOJSON_INCLUDE_DIR "" CACHE PATH "Directory containing ArduinoJson 7.4.3 ArduinoJson.h")
add_library(x2d_gateway INTERFACE)
target_link_libraries(x2d_gateway INTERFACE x2d_core)
if(ARDUINOJSON_INCLUDE_DIR)
  target_include_directories(x2d_gateway INTERFACE ${ARDUINOJSON_INCLUDE_DIR})
endif()

include(CTest)
if(BUILD_TESTING)
  if(NOT EXISTS "${ARDUINOJSON_INCLUDE_DIR}/ArduinoJson.h")
    message(FATAL_ERROR "Tests require ArduinoJson 7.4.3: set ARDUINOJSON_INCLUDE_DIR")
  endif()
  foreach(check IN ITEMS radio radio_runtime protocol cc1101 journal_fixture gateway)
    add_executable(check_${check} tests/check_${check}.cpp)
    if(check STREQUAL "protocol" OR check STREQUAL "gateway")
      target_link_libraries(check_${check} PRIVATE x2d_gateway)
    else()
      target_link_libraries(check_${check} PRIVATE x2d_core)
    endif()
    target_compile_options(check_${check} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    add_test(NAME ${check} COMMAND check_${check})
  endforeach()
  target_compile_definitions(check_journal_fixture PRIVATE
    JOURNAL_FIXTURE="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/journal-v1-a0f98d0.bin")
  if(UNIX)
    add_executable(gateway_server tests/gateway_server.cpp)
    target_link_libraries(gateway_server PRIVATE x2d_gateway)
    target_compile_options(gateway_server PRIVATE -Wall -Wextra -Werror)
  endif()
endif()
